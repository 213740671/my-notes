# 代码
```cpp
#include <bits/stdc++.h>  
using namespace std;  
int main() {  
    //第一行十个数一个一个放进数组vector中  
    vector<int> v;  
    for (int i=0; i<10; i++) {  
        int n;  
        cin >> n;  
        v.push_back(n);  
    }  
    int h;   
    cin >> h;  
    int h_max=h+30;  
    int num=0;  
    for (int i=0; i<10; i++) {  
        if (h_max>=v[i]) {  
            num++;  
        }  
    }  
    cout << num << endl;  
    return 0;  
}
```
