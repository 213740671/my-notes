# 代码
```cpp
#include<bits/stdc++.h>  
using namespace std;  
  
int main() {  
    int x=0,y=0,b=0;  
    cin>>x>>y;  
    int p=x*y;  
    b=x%y;  
    while (b!=0) {  
        x=y;  
        y=b;  
        b=x%y;  
    }  
    cout<<y<<" "<<p/y;  
    return 0;  
}
```
