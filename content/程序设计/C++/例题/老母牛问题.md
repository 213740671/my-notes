# 代码
```cpp
/*老母牛的故事(年初生小母牛)  
 * a 母牛的数量/今年新生小母牛的数量  
 * b0 0岁母牛  
 * b1 1岁母牛  
 * b2 2岁母牛  
 * * 从第一年开始                   总数  
 * a=1 b0=0 b1=0 b2=0  第1年     1 * a=1 b0=1 b1=0 b2=0  第2年     2 * a=1 b0=1 b1=1 b2=0  第3年     3 * a=1 b0=1 b1=1 b2=1  第4年     4 * a=2 b0=2 b1=1 b2=1  第5年     6 */  
  
#include<bits/stdc++.h>  
using namespace std;  
int a=1,b0=0,b1=0,b2=0;  
int sum[60]={0};  
int main() {  
    for (int i=1;i<55;i++) {  
        sum[i]=a+sum[i-1];//总母牛数=去年的母牛数量+今年新出生的母牛数量  
        a=a+b2;//今年老母牛的数量=去年的老母牛+去年2岁今年3岁的小母牛  
        b2=b1;//今年2岁母牛是去年1岁母牛  
        b1=b0;//今年1岁母牛是去年0岁母牛  
        b0=a;//今年0岁母牛是今年年初出生的牛  
    }  
    while (1) {  
        int i;  
        cin >> i;  
        if (i==0)break;  
        cout<<sum[i]<<endl;  
    }  
  
    return 0;  
}
```
