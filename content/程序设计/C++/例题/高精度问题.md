# 大整数加法

1. **输入两个高精度整数**
   - 用 `string s1, s2` 读入两个可能非常大的数字（避免溢出）。

2. **将字符串转换为倒序整数数组**
   - 调用 `strtoint(s, a[])`：
     - 从字符串末尾开始取每一位字符。
     - 转成数字后按倒序存入数组：
       - 最低位存到 `a[1]`，次低位存到 `a[2]` ……  
     - 这样便于从低位到高位逐位相加（模拟手算加法）。

3. **确定结果数组需要的长度**
   - `len1 = s1.size()`，`len2 = s2.size()`
   - `len3 = max(len1, len2) + 1`
     - 多预留 1 位用于处理最高位可能出现的进位。

4. **逐位相加并处理进位**
   - 对 `i = 1 ... len3` 进行循环：
     - `c[i] = a[i] + b[i] + c[i]`（当前位相加 + 上一轮留下的进位）
     - `c[i+1] = c[i] / 10`（计算进位并存到下一位）
     - `c[i] = c[i] % 10`（当前位保留个位数）

5. **去掉最高位多余的 0**
   - 若最高位没有产生进位（`c[len3] == 0`），则 `len3--`，避免输出前导 0。

6. **输出结果**
   - 因为结果数组 `c` 也是倒序存储的，所以从 `len3` 到 `1` 逆序输出，得到最终高精度加法结果。

```cpp
#include <iostream>
using namespace std;

string s1,s2;//高精度数
int a[101],b[101];//存储高精度数转化后的整数的数组
int c[101];//存储结果的数组

  

void strtoint(string s,int a[]){
    int len=s.size();
    for(int i=1;i<=len;i++){
        a[i]=s[len-i]-'0';//将字符串倒序存储到数组中
    }
}



int main(void) {

    cin>>s1>>s2;
    strtoint(s1,a);
    strtoint(s2,b);

    int len1=s1.size();
    int len2=s2.size();
    int len3=max(len1,len2) + 1;

    for(int i=1;i<=len3;i++){
        c[i]=a[i]+b[i]+c[i];//对应位相加   4321
        c[i+1]=c[i]/10;//进位            765
        c[i]=c[i]%10;//当前位取模         1081  
        }

    if(c[len3]==0 & len3 >1){//最高位是否为0
        len3--;
    }

    //输出最后结果
    for(int i=len3;i>=1;i--){
        cout<<c[i];
    }
    
    return 0;

}
```
# 大整数减法

```cpp
#include<bits/stdc++.h>
using namespace std;

//高精度减法
string s1,s2;
int a[210],b[210],c[210];

void toint_reverse(string s,int t[]){
    for(int i=0;i<s.length();i++)
        t[s.length()-i]=s[i]-'0';
}

int main(){
    cin>>s1>>s2;

    //1.将字符串s转换为整型反转（个位对齐）到整型数组中
    toint_reverse(s1,a);  toint_reverse(s2,b);
    int la=s1.length(),lb=s2.length(),lc=max(la,lb);

    //2.对位相减
    for(int i=1/*从个位开始*/;i<=lc;i++){
        if(a[i]<b[i]){
            a[i+1]--;//借位
            a[i]+=10;//借1当10
        }
        c[i]=a[i]-b[i];
    }
    
    //3.去除前导0 （排除000...000+000...000=000...000的情况）
    while(c[lc]==0&&lc>1)lc--;
    
    //4.倒序打印
    for(int i=lc;i>=1;i--) cout<<c[i];
    return 0;
}
```

# 高精度乘法
```cpp
#include <iostream>

#include <string>

#include <vector>

#include <cctype>

  

using namespace std;

  

/*

高精度（大整数）乘法：支持

1) 任意位数（不再用固定长度数组，改用 vector 动态存储）

2) 正负号（+ / -）

3) 前导零处理（如 000123 -> 123；-0 -> 0）

  

实现思路（竖式乘法 / 小学乘法）：

把数字按“低位在前”的顺序存入数组，例如：

  "12345" -> [5,4,3,2,1]

这样相乘时：第 i 位 * 第 j 位 的贡献会累加到 res[i+j]。

  

时间复杂度：O(n*m)，其中 n、m 是两数的位数。

（如果位数非常夸张，需用 FFT/NTT 等更快算法；这里先做“完整通用”的竖式版。）

*/

  

// 去掉字符串表示的前导零（只处理绝对值部分），至少保留 1 位。

static string stripLeadingZeros(const string& digits) {

    size_t pos = 0;

    while (pos + 1 < digits.size() && digits[pos] == '0') {

        ++pos;

    }

    return digits.substr(pos);

}

  

// 把“绝对值数字串（仅包含 0-9）”转换为：低位在前的 digit 数组。

// 例如 "507" -> [7,0,5]

static vector<int> toReversedDigits(const string& digits) {

    vector<int> result;

    result.reserve(digits.size());

    for (int i = static_cast<int>(digits.size()) - 1; i >= 0; --i) {

        result.push_back(digits[i] - '0');

    }

    return result;

}

  

// 解析输入的大整数：处理可选符号位、校验字符、去前导零。

// 返回：

// - negative：是否为负

// - digits：绝对值的“低位在前”数组

// 如果输入非法，则 ok=false。

struct BigIntParsed {

    bool ok = true;

    bool negative = false;

    vector<int> digits; // 低位在前

};

  

static BigIntParsed parseBigInt(const string& s) {

    BigIntParsed parsed;

  

    if (s.empty()) {

        parsed.ok = false;

        return parsed;

    }

  

    size_t idx = 0;

    if (s[0] == '+' || s[0] == '-') {

        parsed.negative = (s[0] == '-');

        idx = 1;

    }

  

    // 符号位后必须还有至少一个数字

    if (idx >= s.size()) {

        parsed.ok = false;

        return parsed;

    }

  

    // 校验必须全是数字

    for (size_t i = idx; i < s.size(); ++i) {

        if (!isdigit(static_cast<unsigned char>(s[i]))) {

            parsed.ok = false;

            return parsed;

        }

    }

  

    // 取出绝对值部分并去掉前导零

    string absDigits = stripLeadingZeros(s.substr(idx));

  

    // "0" 视为非负（统一化：-0 -> 0）

    if (absDigits == "0") {

        parsed.negative = false;

    }

  

    parsed.digits = toReversedDigits(absDigits);

    return parsed;

}

  

// 竖式乘法：输入是低位在前的 digit 数组，输出同样是低位在前。

static vector<int> multiplyAbs(const vector<int>& a, const vector<int>& b) {

    // 任何一个为 0，则结果为 0

    if ((a.size() == 1 && a[0] == 0) || (b.size() == 1 && b[0] == 0)) {

        return vector<int>{0};

    }

  

    // 结果最多有 a.size()+b.size() 位（十进制竖式乘法的常识）

    vector<int> res(a.size() + b.size(), 0);

  

    // 累加每一位的乘积贡献

    for (size_t i = 0; i < a.size(); ++i) {

        for (size_t j = 0; j < b.size(); ++j) {

            res[i + j] += a[i] * b[j];

        }

    }

  

    // 统一处理进位

    int carry = 0;

    for (size_t k = 0; k < res.size(); ++k) {

        int value = res[k] + carry;

        res[k] = value % 10;

        carry = value / 10;

    }

  

    // 理论上这里 carry 应该为 0（因为 res 预留了足够位数），

    // 但为了健壮性，若还有 carry 就继续加到更高位。

    while (carry > 0) {

        res.push_back(carry % 10);

        carry /= 10;

    }

  

    // 去掉结果高位的前导零（对应数组末尾的 0）

    while (res.size() > 1 && res.back() == 0) {

        res.pop_back();

    }

  

    return res;

}

  

// 把低位在前的 digit 数组转回字符串（绝对值部分）

static string digitsToString(const vector<int>& digits) {

    string out;

    out.reserve(digits.size());

    for (int i = static_cast<int>(digits.size()) - 1; i >= 0; --i) {

        out.push_back(static_cast<char>('0' + digits[i]));

    }

    return out;

}

  

int main() {

    ios::sync_with_stdio(false);

    cin.tie(nullptr);

  

    string s1, s2;

    cin >> s1 >> s2;

  

    BigIntParsed x = parseBigInt(s1);

    BigIntParsed y = parseBigInt(s2);

  

    if (!x.ok || !y.ok) {

        // 输入非法时的处理：给出明确提示，方便你调试/复习。

        // 如果你是在 OJ 上提交，一般输入保证合法，你也可以把这段删掉。

        cerr << "Invalid input: please input integers like 123, -456, +789";

        return 1;

    }

  

    // 结果符号：一正一负为负；同号为正。

    bool negativeResult = (x.negative != y.negative);

  

    vector<int> absProduct = multiplyAbs(x.digits, y.digits);

    string absStr = digitsToString(absProduct);

  

    // 结果为 0 时不输出负号

    if (absStr == "0") {

        negativeResult = false;

    }

  

    if (negativeResult) {

        cout << '-';

    }

    cout << absStr;

  

    return 0;

}
```
# 高精度除法(高精度除低精度)
```cpp
#include <bits/stdc++.h>  // 万能头文件（竞赛推荐）

using namespace std;      // 使用标准命名空间

  

string s;

long long b;

int a[201],c[201];

  

void str_to_int(string s,int a[])

{

    for(int i = 0;i < s.length();i++)

    {

        a[i+1]=s[i] - '0';

    }

}

  

int main()

{

    cin >> s ;
    cin >> b ;
    str_to_int(s,a);
    int la = s.length();
    int temp = 0;
    for(int i = 1;i <= la;i++)
    {
        c[i] = (10*temp + a[i]) / b;
        temp = (10*temp + a[i]) % b;
    }

    //去掉前面的无效零
    int lc = 1;
    while(c[lc] == 0 && lc < la) lc++;
    
    //轮询输出结果
    for (int i = lc; i <= (int)s.length(); i++) {
        cout << c[i];
    }
    cout << endl << temp;
}
```
