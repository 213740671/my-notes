# 代码
```cpp
#include<bits/stdc++.h>  
using namespace std;  
/*阶乘函数*/  
long long multiply(int x) {  
    long long sum = 1;  
    for (int i = 1; i <= x; i++) {  
        sum = sum * i;//当X = 4 时,sum=(((1*1)*2)*3)*4  4的阶乘  
    }  
    return sum;  
}  
  
int main() {  
    int a = 0;  
    cin >> a;  
    long long Sn = 0;  
    for (int i = 1; i <= a; i++) {  
        Sn += multiply(i);  
    }  
    cout << Sn << endl;  
    return 0;  
}
```

>[!note]+ 注意变量类型的设置,所有跟和相关的变量都应该是long long
