# 代码
## 使用结构体
```cpp
#include <bits/stdc++.h>  
using namespace std;  
  
struct Student {  
    int num;        //学号是行号减一  
    int chinese;   // 语文成绩  
    int math;  
    int english;  
    int total;  
};  
int main() {  
    int nn;  
    cin>>nn;  
    //将成绩存储好  
    vector<Student> students;  
    for (int i=0;i<nn;i++) {  
        int c;   // 语文成绩  
        int m;  
        int e;  
        cin>>c>>m>>e;  
        students.push_back({.num=i+1,.chinese=c,.math=m,.english=e,.total=m+c+e});  
    }  
    //首先比较总成绩大小,总成绩相同就比较语文成绩,前面都是越大排名越前,最后如果总成绩和语文成绩全部相同就学号越小的在前面  
    sort(students.begin(), students.end(), [](const Student& a, const Student& b) {  
    // 规则 1：总分大的排前面 (降序)  
    if (a.total != b.total) {  
        return a.total > b.total;  
    }  
    // 规则 2：总分相同，语文成绩大的排前面 (降序)  
    if (a.chinese != b.chinese) {  
        return a.chinese > b.chinese;  
    }  
    // 规则 3：前两者都相同，学号小的排前面 (升序)  
    return a.num < b.num;  
    });  
    int count = min((int)students.size(), 5);  
    for (int i = 0; i < count; i++) {  
        cout << students[i].num << " " << students[i].total << endl;  
    }  
    return 0;  
}
```
