# 代码
```cpp
#include<bits/stdc++.h>  
using namespace std;  
  
int main() {  
    int N;  
    cin>>N;  
    int a[N];  
    for(int i=0;i<N;i++) {  
        cin>>a[i];  
    }  
    //a数组里面存着所有的数  
    //排序  
    sort(a,a+N);  
    //删除重复的数,并放在另一个数组中  
    vector<int>b;  
    for (int i=0;i<N;i++) {  
        if (a[i]!=a[i+1]) {  
            b.push_back(a[i]);  
        }  
    }  
    //输出b数组长度和内容  
    cout<<b.size()<<endl;  
    for (int i=0;i<b.size();i++) {  
        cout<<b[i]<<" ";  
    }  
    return 0;  
}
```
