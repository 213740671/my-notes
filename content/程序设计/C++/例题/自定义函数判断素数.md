# 代码
```cpp
#include <bits/stdc++.h>  
using namespace std;  
void IsPrime(int x) {  
    for (int i=2;i<x;i++) {  
        if (x%i==0) {  
            cout<<"not prime"<<endl;  
            return;  
        }  
    }  
    cout<<"prime"<<endl;  
}  
  
int main() {  
    int p;  
    cin>>p;  
    IsPrime(p);  
    return 0;  
}
```
